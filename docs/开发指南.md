## 核心设计原则

### 1. 零硬编码原则

**❌ 错误示例:**
```python
# 硬编码参数
def calculate_runoff(precipitation):
    fc = 300  # 硬编码！
    beta = 2.0  # 硬编码！
    # ...
```

**✅ 正确示例:**
```python
# 从配置加载参数
def calculate_runoff(precipitation, hbv_params):
    fc = hbv_params['FC']
    beta = hbv_params['BETA']
    # ...
```

### 2. 配置文件优先

所有可配置的参数都应该在YAML配置文件中定义：

```yaml
# config/workflow_config.yaml
hbv_parameters:
  FC: 300
  BETA: 2.0
  LP: 0.7

rain_gauge:
  target_density: 0.01
  min_distance_m: 1000
```

### 3. 验证为先

每个关键步骤都应该集成验证：

```python
from hydrosis.validation import ValidationResult

def process_data(data, config):
    # 1. 验证输入
    validation = validate_input_data(data)
    if not validation.is_valid:
        raise ValueError(f"输入验证失败: {validation.errors}")

    # 2. 处理数据
    result = do_processing(data)

    # 3. 验证输出
    validation = validate_output_data(result)
    if not validation.is_valid:
        logger.warning(f"输出验证警告: {validation.warnings}")

    return result, validation
```

### 4. 模块化设计

每个功能模块应该：
- 单一职责
- 独立测试
- 清晰接口

```python
# hydrosis/runoff/hbv.py - HBV模型
# hydrosis/runoff/parallel_hbv.py - HBV并行化
# hydrosis/validation/precipitation.py - 降雨验证
# hydrosis/validation/spatial.py - 空间验证
```

## 验证框架使用

### 基本验证流程

```python
from hydrosis.validation import (
    ValidationResult,
    validate_precipitation_data,
    validate_runoff_coefficient
)

# 1. 创建ValidationResult对象
result = ValidationResult(step_name="Step 5: 降雨分析")

# 2. 添加检查结果
if precipitation < 0:
    result.add_error("降雨不能为负值")

if missing_ratio > 0.1:
    result.add_warning(f"缺测率偏高: {missing_ratio:.2%}")

# 3. 记录关键指标
result.metrics['total_precipitation'] = total_precip
result.metrics['spatial_cv'] = spatial_cv

# 4. 检查验证状态
if result.is_valid:
    print("✓ 验证通过")
else:
    print(f"✗ 验证失败: {result.errors}")
```

### 自定义验证标准

```python
from dataclasses import dataclass
from hydrosis.validation import ValidationCriteria

@dataclass
class MyCustomCriteria(ValidationCriteria):
    """自定义验证标准"""
    max_velocity: float = 5.0  # m/s
    min_depth: float = 0.1     # m
    max_width: float = 100.0   # m

    @classmethod
    def from_dict(cls, data: dict):
        return cls(**{k: v for k, v in data.items()
                     if k in cls.__annotations__})

# 使用自定义标准
criteria = MyCustomCriteria(max_velocity=8.0)
```

### 集成到工作流步骤

```python
def step_05_rainfall_analysis(config, validation_config):
    """Step 5: 降雨分析"""

    # 1. 加载验证标准
    from hydrosis.validation import PrecipitationCriteria
    criteria = PrecipitationCriteria.from_dict(
        validation_config['precipitation']
    )

    # 2. 加载数据
    precip_df = load_precipitation_data(config)

    # 3. 执行验证
    validation = validate_precipitation_data(
        precip_df,
        criteria=criteria,
        step_name="Step 5: 降雨质量检查"
    )

    # 4. 处理验证结果
    print(validation)

    if not validation.is_valid:
        # 保存诊断信息
        save_validation_report(validation, output_dir)
        raise ValueError("降雨数据验证失败")

    # 5. 继续处理...
    return precip_df, validation
```

```python
    """模块配置类

    param2: int = 10
    param3: str = "default"

        """从字典创建配置"""
        return cls(**{k: v for k, v in data.items()
                     if k in cls.__annotations__})


def main_function(
    input_data: pd.DataFrame,
    config: ModuleConfig,
    validate: bool = True
) -> tuple[pd.DataFrame, ValidationResult]:
    """主要功能函数

    Args:
        input_data: 输入数据
        config: 配置对象
        validate: 是否执行验证

    Returns:
        (结果数据, 验证结果)

    Example:
        >>> config = ModuleConfig(param1=2.0)
        >>> result, validation = main_function(data, config)
        >>> print(validation)
    """
    # 1. 输入验证（可选）

        input_validation = _validate_input(input_data)
```

## 性能优化建议

### 1. 使用并行处理

对于大规模计算，使用并行化模块：

```python
from hydrosis.runoff.parallel_hbv import run_hbv_parallel, ParallelHBVConfig

# 自动选择最优worker数量
import multiprocessing
optimal_workers = max(1, multiprocessing.cpu_count() - 1)

config = ParallelHBVConfig(
    max_workers=optimal_workers,
    show_progress=True
)

results = run_hbv_parallel(zones, precip_data, hbv_params, config)
```

### 2. 向量化计算

优先使用NumPy向量化操作：

```python
# ❌ 慢: 循环
runoff = []
for i in range(len(precipitation)):
    r = calculate_single_timestep(precipitation[i])
    runoff.append(r)

# ✅ 快: 向量化
runoff = calculate_vectorized(precipitation)  # NumPy数组操作
```

### 3. 缓存中间结果

```python
from functools import lru_cache

@lru_cache(maxsize=128)
def expensive_calculation(param1, param2):
    """耗时计算，结果会被缓存"""

```

### 4. 惰性加载大数据

```python

all_data = load_entire_dataset()  # 可能很大
```

## 常见问题解决

### 问题1: 径流系数 > 1

**诊断工具:**
```bash
python diagnose_zone2_precipitation.py
```

**常见原因:**
1. 降雨数据空间分布不均（CV > 0.5）
2. 降雨插值方法不当
3. HBV参数不合理

**解决方案:**
```python
# 1. 检查降雨数据质量
from hydrosis.validation import validate_precipitation_data
result = validate_precipitation_data(precip_df)
print(result)

# 2. 查看空间变异系数
spatial_cv = precip_df.std(axis=1).mean() / precip_df.mean(axis=1).mean()
print(f"空间CV: {spatial_cv:.3f}")

# 3. 如果CV过大，改进插值或使用真实数据
```

### 问题2: 验证失败

**示例错误:**
```
ValidationResult: Step 5 降雨验证
状态: ✗ 失败
错误:
  - 空间变异系数过大: 0.65 > 0.50
```

**解决方案:**
```python
# 方案1: 调整验证标准（如果确实合理）
from hydrosis.validation import PrecipitationCriteria

criteria = PrecipitationCriteria(
    max_spatial_cv=0.7  # 放宽标准
)

# 方案2: 改进数据质量
# 使用更多雨量站、改进插值方法等
```

### 问题3: 并行性能差

**诊断:**
```python
from hydrosis.runoff.parallel_hbv import benchmark_parallel_performance

performance = benchmark_parallel_performance(
    zones, precip_data, hbv_params,
    worker_counts=[1, 2, 4, 8]
)

for workers, time in performance.items():
    print(f"{workers} workers: {time:.2f}秒")
```

**优化建议:**
- 分区数 < 4: 使用串行（max_workers=1）
- 分区数 4-20: max_workers=2-4
- 分区数 > 20: max_workers=4-8

### 问题4: 配置文件路径错误

**常见错误:**
```python
FileNotFoundError: config/workflow_config.yaml
```

**解决方案:**
```python
from pathlib import Path

# 使用绝对路径
project_root = Path(__file__).parent.parent
config_path = project_root / 'config' / 'workflow_config.yaml'

# 或者使用环境变量
import os
config_path = os.environ.get('HYDROSIS_CONFIG', 'config/workflow_config.yaml')
```

## 测试规范

### 单元测试

```python
# test_hbv.py
import unittest
import numpy as np
from hydrosis.runoff.hbv import calculate_runoff

class TestHBV(unittest.TestCase):
    def setUp(self):
        """测试前准备"""
        self.precip = np.array([0, 5, 10, 8, 3, 0])
        self.hbv_params = {
            'FC': 300, 'BETA': 2.0, 'LP': 0.7,
            'K0': 0.1, 'K1': 0.05, 'K2': 0.01,
            'PERC': 2.0, 'UZL': 50.0
        }

    def test_runoff_coefficient_range(self):
        """测试径流系数在合理范围内"""
        result = calculate_runoff(self.precip, 100.0, self.hbv_params)
        rc = result['runoff_coefficient']
        self.assertGreaterEqual(rc, 0.0)
        self.assertLessEqual(rc, 1.0)

    def test_zero_precipitation(self):
        """测试零降雨情况"""
        zero_precip = np.zeros(10)
        result = calculate_runoff(zero_precip, 100.0, self.hbv_params)
        self.assertEqual(result['total_runoff_mm'], 0.0)

if __name__ == '__main__':
    unittest.main()
```
