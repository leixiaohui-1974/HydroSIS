#### 8.1.2 自动化部署脚本

```bash
#!/bin/bash
# deploy_aliyun.sh - 阿里云自动化部署脚本

set -e

# 配置参数
REGION="cn-hangzhou"
PROJECT_NAME="hydrosis"
ENV="production"

echo "==> 开始部署HydroSIS云服务 (环境: $ENV)"

# 1. 构建Docker镜像
echo "==> 构建Docker镜像..."
docker build -t registry.cn-hangzhou.aliyuncs.com/${PROJECT_NAME}/app:latest \
  -f Dockerfile.app .

docker build -t registry.cn-hangzhou.aliyuncs.com/${PROJECT_NAME}/worker:latest \
  -f Dockerfile.worker .

# 2. 推送到阿里云镜像仓库
echo "==> 推送镜像到ACR..."
docker push registry.cn-hangzhou.aliyuncs.com/${PROJECT_NAME}/app:latest
docker push registry.cn-hangzhou.aliyuncs.com/${PROJECT_NAME}/worker:latest

# 3. 更新ECS实例
echo "==> 更新ECS实例..."
for instance_id in $(aliyun ecs DescribeInstances \
  --RegionId $REGION \
  --Tag.1.Key=Project \
  --Tag.1.Value=$PROJECT_NAME \
  --query 'Instances.Instance[].InstanceId' \
  --output text); do
  
  echo "  更新实例: $instance_id"
  aliyun ecs RunCommand \
    --RegionId $REGION \
    --InstanceId.1 $instance_id \
    --CommandContent "$(cat <<'EOF'
#!/bin/bash
docker pull registry.cn-hangzhou.aliyuncs.com/hydrosis/app:latest
docker stop hydrosis-app || true
docker rm hydrosis-app || true
docker run -d \
  --name hydrosis-app \
  --restart always \
  -p 8000:8000 \
  -e DATABASE_URL=${DATABASE_URL} \
  -e REDIS_URL=${REDIS_URL} \
  -e OSS_ENDPOINT=${OSS_ENDPOINT} \
  registry.cn-hangzhou.aliyuncs.com/hydrosis/app:latest
EOF
)" \
    --Type RunShellScript \
    --Timeout 300
done

# 4. 更新函数计算
echo "==> 更新函数计算..."
cd functions/auth
aliyun fc UpdateFunction \
  --serviceName ${PROJECT_NAME}-services \
  --functionName auth-handler \
  --codeDir .

cd ../chat
aliyun fc UpdateFunction \
  --serviceName ${PROJECT_NAME}-services \
  --functionName chat-handler \
  --codeDir .

# 5. 更新前端静态资源到OSS
echo "==> 更新前端资源..."
cd ../../frontend
npm run build

ossutil cp -r dist/ oss://hydrosis-public/web/ \
  --update \
  --recursive

# 6. 刷新CDN缓存
echo "==> 刷新CDN..."
aliyun cdn RefreshObjectCaches \
  --ObjectPath https://www.hydrosis.com/ \
  --ObjectType Directory

# 7. 健康检查
echo "==> 执行健康检查..."
for i in {1..10}; do
  if curl -f https://api.hydrosis.com/health; then
    echo "✓ 健康检查通过"
    break
  else
    echo "  等待服务启动... ($i/10)"
    sleep 10
  fi
done

echo "==> 部署完成！"
```

#### 8.2.1 Docker镜像

```dockerfile
# Dockerfile.mcp-server

FROM python:3.10-slim

# 安装系统依赖
RUN apt-get update && apt-get install -y \
    gcc g++ \
    libgdal-dev \
    libgeos-dev \
    libproj-dev \
    git \
    && rm -rf /var/lib/apt/lists/*

# 设置工作目录
WORKDIR /app

# 复制依赖文件
COPY requirements.txt .
COPY requirements-mcp.txt .

# 安装Python依赖
RUN pip install --no-cache-dir -r requirements.txt \
    && pip install --no-cache-dir -r requirements-mcp.txt

# 复制应用代码
COPY hydrosis/ /app/hydrosis/
COPY mcp_server/ /app/mcp_server/
COPY config/ /app/config/

# 创建数据目录
RUN mkdir -p /data/projects /data/cache /data/logs

# 暴露端口
EXPOSE 8080

# 健康检查
HEALTHCHECK --interval=30s --timeout=10s --start-period=40s --retries=3 \
  CMD python -c "import requests; requests.get('http://localhost:8080/health')"

# 启动命令
CMD ["uvicorn", "mcp_server.main:app", "--host", "0.0.0.0", "--port", "8080", "--workers", "4"]
```

#### 8.2.3 部署脚本

```bash
#!/bin/bash
# deploy_mcp.sh - MCP服务部署脚本

set -e

NAMESPACE="hydrosis"
VERSION="v1.0.0"

echo "==> 开始部署HydroSIS MCP服务 (版本: $VERSION)"

# 1. 创建命名空间
kubectl create namespace $NAMESPACE --dry-run=client -o yaml | kubectl apply -f -

# 2. 创建配置
kubectl apply -f k8s/configmap.yaml
kubectl apply -f k8s/secrets.yaml

# 3. 创建存储
kubectl apply -f k8s/pvc.yaml

# 4. 部署PostgreSQL
helm install postgres bitnami/postgresql \
  --namespace $NAMESPACE \
  --set auth.username=hydrosis \
  --set auth.password=<密码> \
  --set auth.database=hydrosis \
  --set primary.persistence.size=100Gi

# 5. 部署Redis
helm install redis bitnami/redis \
  --namespace $NAMESPACE \
  --set auth.enabled=false \
  --set master.persistence.size=10Gi

# 6. 部署MinIO
helm install minio bitnami/minio \
  --namespace $NAMESPACE \
  --set auth.rootUser=admin \
  --set auth.rootPassword=<密码> \
  --set defaultBuckets=hydrosis-projects,hydrosis-results \
  --set persistence.size=1Ti

# 7. 构建并推送镜像
docker build -t hydrosis/mcp-server:$VERSION -f Dockerfile.mcp-server .
docker build -t hydrosis/worker:$VERSION -f Dockerfile.worker .

docker push hydrosis/mcp-server:$VERSION
docker push hydrosis/worker:$VERSION

# 8. 部署MCP服务
kubectl apply -f k8s/deployment-mcp.yaml
kubectl apply -f k8s/service-mcp.yaml

# 9. 部署Worker
kubectl apply -f k8s/deployment-worker.yaml

# 10. 配置自动伸缩
kubectl apply -f k8s/hpa.yaml

# 11. 配置Ingress
kubectl apply -f k8s/ingress.yaml

# 12. 等待部署完成
kubectl rollout status deployment/hydrosis-mcp-server -n $NAMESPACE --timeout=300s
kubectl rollout status deployment/hydrosis-worker -n $NAMESPACE --timeout=300s

# 13. 验证部署
echo "==> 验证部署状态..."
kubectl get pods -n $NAMESPACE
kubectl get svc -n $NAMESPACE

# 14. 健康检查
MCP_URL=$(kubectl get svc hydrosis-mcp-service -n $NAMESPACE -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
echo "==> MCP服务地址: http://$MCP_URL"

if curl -f http://$MCP_URL/health; then
  echo "✓ 健康检查通过"
else
  echo "✗ 健康检查失败"
  exit 1
fi

echo "==> 部署完成！"
```

          severity: warning

          summary: "请求延迟过高"

        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "错误率过高"
          description: "错误率超过5%"
      
      # 任务队列积压

        expr: celery_queue_length{queue="simulation"} > 100
        for: 15m
        labels:
          severity: warning
        annotations:

### 8.4 灾备与恢复

#### 8.4.1 备份策略

```bash
#!/bin/bash
# backup.sh - 自动备份脚本

BACKUP_DIR="/backup/hydrosis/$(date +%Y%m%d)"
mkdir -p $BACKUP_DIR

echo "==> 开始备份 ($(date))"

# 1. 数据库备份
echo "  备份PostgreSQL数据库..."
pg_dump -h postgres -U hydrosis -d hydrosis | gzip > $BACKUP_DIR/database.sql.gz

# 2. MinIO数据备份
echo "  备份MinIO对象存储..."
mc mirror minio/hydrosis-projects $BACKUP_DIR/minio/projects
mc mirror minio/hydrosis-results $BACKUP_DIR/minio/results

# 3. 配置备份
echo "  备份Kubernetes配置..."
kubectl get all,configmap,secret,pvc -n hydrosis -o yaml > $BACKUP_DIR/k8s-config.yaml

# 4. 上传到OSS
echo "  上传备份到阿里云OSS..."
ossutil cp -r $BACKUP_DIR oss://hydrosis-backup/$(date +%Y%m%d)/

# 5. 清理本地旧备份（保留7天）
find /backup/hydrosis -type d -mtime +7 -exec rm -rf {} \;

# 6. 验证备份完整性
echo "  验证备份..."
gzip -t $BACKUP_DIR/database.sql.gz && echo "✓ 数据库备份验证通过"

echo "==> 备份完成 ($(date))"
```

#### 8.4.2 恢复流程

```bash
#!/bin/bash
# restore.sh - 数据恢复脚本

BACKUP_DATE=$1
BACKUP_DIR="/backup/hydrosis/$BACKUP_DATE"

if [ -z "$BACKUP_DATE" ]; then
  echo "用法: $0 YYYYMMDD"
  exit 1
fi

echo "==> 开始恢复 $BACKUP_DATE 的备份"

# 1. 从OSS下载备份
echo "  下载备份文件..."
ossutil cp -r oss://hydrosis-backup/$BACKUP_DATE/ $BACKUP_DIR

# 2. 恢复数据库
echo "  恢复数据库..."
kubectl scale deployment hydrosis-mcp-server -n hydrosis --replicas=0  # 停止应用
gunzip < $BACKUP_DIR/database.sql.gz | psql -h postgres -U hydrosis -d hydrosis

# 3. 恢复MinIO数据
echo "  恢复对象存储..."
mc mirror $BACKUP_DIR/minio/projects minio/hydrosis-projects
mc mirror $BACKUP_DIR/minio/results minio/hydrosis-results

# 4. 恢复Kubernetes配置（可选）
# kubectl apply -f $BACKUP_DIR/k8s-config.yaml

# 5. 重启应用
echo "  重启应用..."
kubectl scale deployment hydrosis-mcp-server -n hydrosis --replicas=3

# 6. 验证
echo "  验证服务状态..."
kubectl rollout status deployment/hydrosis-mcp-server -n hydrosis

echo "==> 恢复完成"
```
