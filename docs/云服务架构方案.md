## 1. 项目背景与目标

### 1.1 项目背景

HydroSIS 是一个功能完善的分布式水文模拟框架，具备：
- 完整的水文建模能力（产流、汇流、参数分区）
- 多种产流算法（SCS, XinAnJiang, HBV, HYMOD, VIC等）
- 工作流引擎和评价系统
- REST API 和门户系统
- 参数校准和不确定性分析

### 1.2 核心需求

1. **自然语言交互**：用户通过类似 Claude 的对话界面进行水文分析
2. **多用户隔离**：每个用户有独立的账户空间管理数据和对话历史
3. **大规模并发**：支持数千用户同时访问和计算
4. **国内优化**：基于阿里云服务，符合国内合规要求
5. **架构解耦**：通用大模型服务（阿里云）+ 专业计算服务（本地）

### 1.3 设计原则

- **前后端分离**：大模型入口与计算引擎解耦
- **微服务架构**：每个功能模块独立部署和扩展
- **标准协议**：采用 MCP（Model Context Protocol）实现服务互通
- **云原生设计**：容器化部署，支持弹性伸缩
- **数据安全**：用户数据隔离，敏感计算在私有服务器

##### B. 对话服务 (chat-service)
```python
# services/chat/handler.py

from typing import AsyncGenerator
from dashscope import Generation  # 通义千问SDK
from mcp import Client as MCPClient

class ChatService:
    def __init__(self):
        self.qwen_client = Generation()
        self.mcp_client = MCPClient(base_url=os.environ['MCP_SERVER_URL'])
        
    async def process_message(
        self, 
        user_id: str,
        conversation_id: str,
        message: str
    ) -> AsyncGenerator[str, None]:
        """处理用户消息（流式响应）"""
        
        # 1. 加载对话历史
        history = await self.load_conversation(user_id, conversation_id)
        
        # 2. 构建系统提示词
        system_prompt = self._build_system_prompt()
        
        # 3. 获取可用工具列表（从MCP服务器）
        tools = await self.mcp_client.list_tools()
        
        # 4. 调用通义千问（支持工具调用）
        messages = history + [{"role": "user", "content": message}]
        
        response = await self.qwen_client.call(
            model='qwen-max',
            messages=messages,
            tools=self._convert_mcp_tools(tools),
            stream=True,
            result_format='message'
        )
        
        # 5. 处理响应
        async for chunk in response:
            # 如果是工具调用
            if chunk.get('tool_calls'):
                for tool_call in chunk['tool_calls']:
                    # 调用MCP服务器执行工具
                    result = await self.mcp_client.call_tool(
                        tool_call['name'],
                        tool_call['arguments']
                    )
                    # 将结果返回给LLM
                    yield self._format_tool_result(result)
            else:
                # 普通文本响应
                yield chunk['content']
        
        # 6. 保存对话历史
        await self.save_conversation(user_id, conversation_id, messages, response)
    
    def _build_system_prompt(self) -> str:
        """构建系统提示词"""
        return """你是HydroSIS水文模拟助手，专门帮助用户进行水文分析和建模。

你可以使用以下工具：
- create_project: 创建水文模拟项目
- upload_dem: 上传DEM数据
- delineate_watershed: 进行流域划分
- configure_model: 配置水文模型参数
- run_simulation: 运行模拟
- analyze_results: 分析结果

请用简洁、专业的中文回答用户问题。当需要执行计算任务时，主动调用相应工具。
"""
    
    def _convert_mcp_tools(self, mcp_tools: list) -> list:
        """将MCP工具转换为通义千问工具格式"""
        return [
            {
                "type": "function",
                "function": {
                    "name": tool['name'],
                    "description": tool['description'],
                    "parameters": tool['inputSchema']
                }
            }
            for tool in mcp_tools
        ]
```

```python
# services/llm/qwen_client.py

from dashscope import Generation
from typing import List, Dict, AsyncGenerator

class QwenClient:
    def __init__(self, api_key: str):
        self.api_key = api_key
        self.model = 'qwen-max'  # 或 qwen-plus
    
    async def chat_stream(
        self,
        messages: List[Dict[str, str]],
        tools: List[Dict] = None,
        temperature: float = 0.7,
    ) -> AsyncGenerator[Dict, None]:
        """流式对话"""
        
        response = Generation.call(
            api_key=self.api_key,
            model=self.model,
            messages=messages,
            tools=tools,
            stream=True,
            temperature=temperature,
            top_p=0.8,
            result_format='message',
            incremental_output=True
        )
        
        for chunk in response:
            if chunk.status_code == 200:
                yield {
                    'type': 'text' if not chunk.output.get('tool_calls') else 'tool_call',
                    'content': chunk.output.get('text', ''),
                    'tool_calls': chunk.output.get('tool_calls', []),
                    'finish_reason': chunk.output.get('finish_reason')
                }
            else:
                raise Exception(f'通义千问API错误: {chunk.message}')
```

#### 3.2.2 工具调用流程

```python
# 工具调用完整流程
async def process_with_tools(user_message: str):
    """带工具调用的完整流程"""
    
    # 1. 构建消息
    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": user_message}
    ]
    
    # 2. 获取MCP工具定义
    mcp_tools = await mcp_client.list_tools()
    qwen_tools = convert_to_qwen_format(mcp_tools)
    
    # 3. 首次调用LLM
    response = await qwen_client.chat_stream(messages, tools=qwen_tools)
    
    # 4. 处理工具调用
    assistant_message = {"role": "assistant", "content": "", "tool_calls": []}
    
    async for chunk in response:
        if chunk['type'] == 'text':
            assistant_message['content'] += chunk['content']
            yield {'type': 'text', 'content': chunk['content']}
        
        elif chunk['type'] == 'tool_call':
            # 执行工具调用
            for tool_call in chunk['tool_calls']:
                tool_name = tool_call['function']['name']
                tool_args = json.loads(tool_call['function']['arguments'])
                
                # 调用MCP服务器
                tool_result = await mcp_client.call_tool(tool_name, tool_args)
                
                # 返回工具执行结果
                yield {
                    'type': 'tool_result',
                    'tool_name': tool_name,
                    'result': tool_result
                }
                
                # 记录工具调用
                assistant_message['tool_calls'].append({
                    'id': tool_call['id'],
                    'type': 'function',
                    'function': tool_call['function']
                })
                
                # 添加工具结果到消息历史
                messages.append(assistant_message)
                messages.append({
                    "role": "tool",
                    "tool_call_id": tool_call['id'],
                    "content": json.dumps(tool_result, ensure_ascii=False)
                })
        
        # 5. 如果有工具调用，再次调用LLM获取最终回答
        if assistant_message.get('tool_calls'):
            final_response = await qwen_client.chat_stream(messages)
            async for chunk in final_response:
                yield {'type': 'text', 'content': chunk['content']}
```

```python
            'concurrent_tasks': 5                  # 5个并发任务
        }
        
        current_usage = await self.get_usage(user_id, resource_type)
        return current_usage < quotas.get(resource_type, 0)
    
    async def increment_usage(self, user_id: str, resource_type: str, amount: int):
        """增加使用量"""
        key = f'quota:{user_id}:{resource_type}'
        await self.redis.incrby(key, amount)
        
        # 设置过期时间
        if resource_type == 'api_calls_daily':
            await self.redis.expire(key, 86400)  # 24小时
```

```python
def verify_request_signature(request: Request):

    signature = request.headers.get('X-Signature')
    timestamp = request.headers.get('X-Timestamp')
    
    # 防重放攻击
```
